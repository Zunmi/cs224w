\section{Relational Classification I}

Let
\[
  p_i^{(t)}=P(Y_i=+)
\]
denote the probability for node $i$ after iteration $t$. The labeled nodes
are clamped to their ground-truth values during propagation:
\[
  p_5^{(t)}=0,\qquad
  p_6^{(t)}=p_7^{(t)}=p_{10}^{(t)}=1.
\]
All other nodes are initialized to $1/2$. From the graph,
\begin{align*}
  \mathcal{N}_1&=\{2,3\},&
  \mathcal{N}_2&=\{1,3,4,5\},&
  \mathcal{N}_3&=\{1,2,6\},\\
  \mathcal{N}_4&=\{2,5,7,8\},&
  \mathcal{N}_8&=\{4,9\},&
  \mathcal{N}_9&=\{5,8\}.
\end{align*}

The updates are in-place and follow ascending node ID. Thus, within one
iteration, a node uses the new value of a lower-ID node that has already been
updated and the old value of a higher-ID node that has not yet been updated.
The first iteration is
\begin{align*}
  p_1^{(1)}
    &=\frac{p_2^{(0)}+p_3^{(0)}}{2}
      =\frac{1}{2},\\
  p_2^{(1)}
    &=\frac{p_1^{(1)}+p_3^{(0)}+p_4^{(0)}+p_5}{4}
      =\frac{3}{8},\\
  p_3^{(1)}
    &=\frac{p_1^{(1)}+p_2^{(1)}+p_6}{3}
      =\frac{5}{8},\\
  p_4^{(1)}
    &=\frac{p_2^{(1)}+p_5+p_7+p_8^{(0)}}{4}
      =\frac{15}{32},\\
  p_8^{(1)}
    &=\frac{p_4^{(1)}+p_9^{(0)}}{2}
      =\frac{31}{64},\\
  p_9^{(1)}
    &=\frac{p_5+p_8^{(1)}}{2}
      =\frac{31}{128}.
\end{align*}
Using these values, the second iteration is
\begin{align*}
  p_1^{(2)}
    &=\frac{p_2^{(1)}+p_3^{(1)}}{2}
      =\frac{1}{2},\\
  p_2^{(2)}
    &=\frac{p_1^{(2)}+p_3^{(1)}+p_4^{(1)}+p_5}{4}
      =\frac{51}{128},\\
  p_3^{(2)}
    &=\frac{p_1^{(2)}+p_2^{(2)}+p_6}{3}
      =\frac{81}{128},\\
  p_4^{(2)}
    &=\frac{p_2^{(2)}+p_5+p_7+p_8^{(1)}}{4}
      =\frac{241}{512},\\
  p_8^{(2)}
    &=\frac{p_4^{(2)}+p_9^{(1)}}{2}
      =\frac{365}{1024},\\
  p_9^{(2)}
    &=\frac{p_5+p_8^{(2)}}{2}
      =\frac{365}{2048}.
\end{align*}

\subsection{\texorpdfstring{$P(Y_3=+)$}{P(Y3=+)}}

\begin{solution}
\[
  \boxed{
    P(Y_3=+)=p_3^{(2)}
    =\frac{81}{128}
    =0.6328125
  }.
\]
\end{solution}

\subsection{\texorpdfstring{$P(Y_4=+)$}{P(Y4=+)}}

    =\frac{241}{512}
    =0.470703125
  }.
\]
\end{solution}

\subsection{Predicted Negative-Class Nodes}

\begin{solution}
A node is assigned to class $+$ only when $P(Y_i=+)>0.5$. After the
second iteration, node 3 and the fixed positive nodes 6, 7, and 10 are in
class $+$. Node 1 has probability exactly $0.5$, so the strict threshold
places it in class $-$, together with nodes 2, 4, 5, 8, and 9. In the
required format, the answer is
\[
  \boxed{\texttt{1,2,4,5,8,9}}.
\]
\end{solution}
