\section{Link Analysis}

\subsection{Personalized PageRank I}

Eloise has teleport set $\{2\}$. Determine whether her personalized PageRank
vector can be computed from the known vectors $v_A,v_B,v_C,v_D$. If it can,
express it in terms of those vectors; otherwise, explain why.

\begin{solution}
Let $\vect{s}$ denote a teleport probability vector and let
$\vect{v}(\vect{s})$ be its personalized PageRank vector. For a fixed graph
and a fixed teleport parameter $\beta$,
\[
  \vect{v}(\vect{s})
  = \beta M\vect{v}(\vect{s})+(1-\beta)\vect{s},
\]
so
\[
  \vect{v}(\vect{s})
  = (1-\beta)(I-\beta M)^{-1}\vect{s}.
\]
Thus, personalized PageRank is a linear function of the teleport vector.

Let $\vect{e}_i$ be the vector that is one at node $i$ and zero elsewhere.
The known teleport vectors are
\begin{align*}
  \vect{s}_A &= \frac{1}{3}(\vect{e}_1+\vect{e}_2+\vect{e}_3),&
  \vect{s}_B &= \frac{1}{3}(\vect{e}_3+\vect{e}_4+\vect{e}_5),\\
  \vect{s}_C &= \frac{1}{3}(\vect{e}_1+\vect{e}_4+\vect{e}_5),&
  \vect{s}_D &= \vect{e}_1.
\end{align*}
The teleport vector for Eloise can be written as
\begin{align*}
  3\vect{s}_A-3\vect{s}_B+3\vect{s}_C-2\vect{s}_D
  &= (\vect{e}_1+\vect{e}_2+\vect{e}_3)
     -(\vect{e}_3+\vect{e}_4+\vect{e}_5)\\
  &\quad+(\vect{e}_1+\vect{e}_4+\vect{e}_5)-2\vect{e}_1\\
  &= \vect{e}_2.
\end{align*}
Applying the same linear combination to the corresponding PageRank vectors
therefore gives
\[
  \boxed{\vect{v}_{E}=3\vect{v}_A-3\vect{v}_B
  +3\vect{v}_C-2\vect{v}_D.}
\]
\end{solution}

\subsection{Personalized PageRank I (continued)}

Felicity has teleport set $\{5\}$. Answer the same question as in 1.1.

\begin{solution}
The teleport vector for Felicity is $\vect{s}_F=\vect{e}_5$. Consider an
arbitrary linear combination of the known teleport vectors:
\[
  \vect{s}=a\vect{s}_A+b\vect{s}_B+c\vect{s}_C+d\vect{s}_D.
\]
In every known teleport vector, the entries for nodes 4 and 5 are equal.
Consequently, every such linear combination must satisfy
\[
  s_4=s_5=\frac{b+c}{3}.
\]
However, $\vect{e}_5$ has $(s_4,s_5)=(0,1)$, so
\[
  \vect{e}_5\notin
  \operatorname{span}\{\vect{s}_A,\vect{s}_B,\vect{s}_C,\vect{s}_D\}.
\]
Therefore, the known vectors cannot separate node 5 from node 4. Without
accessing the web graph, the personalized PageRank vector for Felicity
\[
  \boxed{\vect{v}_F\text{ cannot be computed from }
  \vect{v}_A,\vect{v}_B,\vect{v}_C,\vect{v}_D.}
\]
\end{solution}

\subsection{Weighted Personalized PageRank}

Glynnis has teleport set $\{1,2,3,4,5\}$ with weights
\[
  \vect{w} = [0.1,\,0.2,\,0.3,\,0.2,\,0.2].
\]
Determine whether her personalized PageRank vector can be computed from
$v_A,v_B,v_C,v_D$ and justify the result.

\begin{solution}
Let $\vect{s}_G$ be the teleport vector for Glynnis. It has the decomposition
\[
  \vect{s}_G
  =0.6\vect{s}_A+0.3\vect{s}_B+0.3\vect{s}_C-0.2\vect{s}_D.
\]
Indeed, expanding the right-hand side gives
\begin{align*}
  &0.2(\vect{e}_1+\vect{e}_2+\vect{e}_3)
  +0.1(\vect{e}_3+\vect{e}_4+\vect{e}_5)\\
  &\quad+0.1(\vect{e}_1+\vect{e}_4+\vect{e}_5)
  -0.2\vect{e}_1\\
  &=0.1\vect{e}_1+0.2\vect{e}_2+0.3\vect{e}_3
    +0.2\vect{e}_4+0.2\vect{e}_5,
\end{align*}
which is exactly the required weighted teleport vector. By linearity,
\[
  \boxed{\vect{v}_G
  =0.6\vect{v}_A+0.3\vect{v}_B+0.3\vect{v}_C-0.2\vect{v}_D.}
\]
The negative coefficient does not cause a problem: the intermediate linear
combination need not be a convex combination, while the resulting teleport
vector $\vect{s}_G$ is still a valid probability vector with nonnegative
entries that sum to one.
\end{solution}

\subsection{Personalized PageRank II}

Given a set $V$ of already-computed personalized PageRank vectors,
characterize every personalized PageRank vector that can be computed from
$V$ without accessing the web graph.

\begin{solution}
Let the known personalized PageRank vectors be
$V=\{\vect{v}_1,\ldots,\vect{v}_m\}$, and let their corresponding teleport
vectors be $\vect{s}_1,\ldots,\vect{s}_m$. Define the probability simplex
\[
  \Delta^{N-1}
  =\left\{\vect{s}\in\R^N:
    \vect{s}\geq 0,\ \vect{1}^{\mathsf T}\vect{s}=1\right\}.
\]
As shown in 1.1, all the PageRank vectors use the same linear map
\[
  T=(1-\beta)(I-\beta M)^{-1},
  \qquad \vect{v}_i=T\vect{s}_i.
\]
Therefore, the complete set that can be computed without accessing the graph
is
\[
  \boxed{
  \mathcal{C}
  =
  \left\{
    \sum_{i=1}^{m}\alpha_i\vect{v}_i:
    \sum_{i=1}^{m}\alpha_i\vect{s}_i\in\Delta^{N-1}
  \right\}
  =
  T\left(
    \operatorname{span}\{\vect{s}_1,\ldots,\vect{s}_m\}
    \cap\Delta^{N-1}
  \right).
  }
\]
Indeed, whenever
$\vect{s}=\sum_i\alpha_i\vect{s}_i$ is a valid teleport vector, linearity
gives
\[
  \vect{v}(\vect{s})
  =T\left(\sum_i\alpha_i\vect{s}_i\right)
  =\sum_i\alpha_iT\vect{s}_i
  =\sum_i\alpha_i\vect{v}_i.
\]
The coefficients need not all be nonnegative, as demonstrated in 1.1 and
1.3; only the resulting teleport vector must be a valid probability vector.
Convex combinations are always valid, but they do not describe the entire
set in general.
\end{solution}

\subsection{A Different Equation for PageRank}

Prove that
\[
  \vect{r}=A\vect{r},
  \qquad
  A=\beta M+\frac{1-\beta}{N}\vect{1}\vect{1}^{\mathsf T}
\]
is equivalent to
\[
  \vect{r}=\beta M\vect{r}+\frac{1-\beta}{N}\vect{1}.
\]
Use the normalization of the PageRank vector explicitly.

\begin{solution}
Starting from (1) and substituting the definition of $A$ gives
\begin{align*}
  \vect{r}
  =A\vect{r}
  &=\left(
      \beta M+\frac{1-\beta}{N}
      \vect{1}\vect{1}^{\mathsf T}
    \right)\vect{r}\\
  &=\beta M\vect{r}
    +\frac{1-\beta}{N}
      \vect{1}\left(\vect{1}^{\mathsf T}\vect{r}\right).
\end{align*}
Because the PageRank vector is normalized,
\[
  \vect{1}^{\mathsf T}\vect{r}
  =\sum_{i=1}^{N}r_i=1.
\]
Hence,
\[
  \vect{r}
  =\beta M\vect{r}
   +\frac{1-\beta}{N}\vect{1},
\]
which is (4).

Conversely, suppose that the normalized vector $\vect{r}$ satisfies (4).
Then
\begin{align*}
  A\vect{r}
  &=\beta M\vect{r}
    +\frac{1-\beta}{N}
      \vect{1}\left(\vect{1}^{\mathsf T}\vect{r}\right)\\
  &=\beta M\vect{r}
    +\frac{1-\beta}{N}\vect{1}\\
  &=\vect{r}.
\end{align*}
Thus (4) also implies (1), so the two PageRank equations are equivalent.
\end{solution}
